# Point Docker CLI to Minikube's Docker daemon
& minikube -p minikube docker-env | Invoke-Expression

# Build your backend and frontend images
docker build -t local-backend .
docker build -t local-frontend .

#Enable ingress
minikube addons enable ingress

kubectl apply -f .\k8s\

 kubectl set image deployment/frontend frontend=local-frontend:v1.0.1

 kubectl get svc backend-service -n frontend -o wide

kubectl port-forward svc/frontend-service 4200:80

minikube service frontend-service

minikube stop
minikube delete --all --purge
docker system prune -a --volumes
minikube start --driver=docker

kubectl delete pod load-generator

sudo usermod -aG docker $USER
newgrp docker

./build-and-deploy.sh

./load-generator.sh
